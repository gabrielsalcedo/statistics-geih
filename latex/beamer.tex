\begin{frame}{Pregunta}
  De cuatro estimadores candidatos del índice de Theil, ¿cuál tiene mejores
  propiedades estadísticas (menor sesgo, menor ECM y consistencia) para estimar
  la desigualdad del ingreso laboral a partir de la GEIH?
  \vspace{1em}

  Con el estimador elegido: ¿cómo cambió la desigualdad entre 2018 y 2026, y qué
  la explica?
\end{frame}

\begin{frame}{Cuatro estimadores del índice de Theil}
  \begin{itemize}
    \item A: con factor de expansión
    \item B: sin factor de expansión
    \item C: lognormal por momentos
    \item D: lognormal por máxima verosimilitud
  \end{itemize}
\end{frame}

\begin{frame}{Hipótesis}
  \begin{itemize}
    \item \textbf{H1.} El estimador ponderado (A) es consistente y tiene el menor ECM cuando $n$ crece.
    \item \textbf{H2.} En 2026 la educación explica más desigualdad que cada una de las otras dimensiones.
    \item \textbf{H3.} La desigualdad del ingreso laboral fue menor en 2026 que en 2018.
  \end{itemize}
\end{frame}

\begin{frame}{Datos: GEIH 2018--2026, enero a junio}
  \begin{itemize}
    \item Ocupados de 15 años o más, remunerados, con ingreso positivo, en 24 departamentos.
    \item Entre 130.000 y 161.000 personas por año; 2020 no entra (recolección telefónica).
    \item Factor de expansión ajustado por no respuesta de ingreso.
    \item Cinco dimensiones: educación, formalidad, estrato, región y sexo.
  \end{itemize}
\end{frame}

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item \textbf{H1 se cumple:} A es el único consistente; con $n = 50.000$ su ECM es entre 4 y 24 veces menor. B y D subestiman la desigualdad en cerca de 15~\%.
    \item \textbf{H2 no se cumple:} en 2026 el estrato explica más (35~\%) y la educación (25~\%) empata con la formalidad (25~\%).
    \item \textbf{H3 no se cumple:} el Theil bajó de 0,439 (2018) a 0,429 (2026), pero la diferencia no es significativa.
  \end{itemize}
\end{frame}
