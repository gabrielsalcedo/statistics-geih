\section{Problema y pregunta}

\begin{frame}{¿Por qué importa cómo se estima la desigualdad?}
  \begin{itemize}
    \item Colombia es uno de los países más desiguales de América Latina.
    \item Las cifras de desigualdad salen de \textbf{encuestas}: son estimaciones, no la desigualdad ``real''.
    \item Con los mismos datos hay varias formas de estimar el índice de Theil, y elegir mal puede producir \textbf{sesgo}.
  \end{itemize}
  \vspace{1em}
  \begin{block}{Pregunta}
    De cuatro estimadores del índice de Theil, ¿cuál tiene mejores propiedades (menor sesgo, menor ECM, consistencia) con la GEIH? Y con ese estimador: ¿cómo cambió la desigualdad del ingreso laboral entre 2018 y 2026, y qué la explica?
  \end{block}
\end{frame}

\begin{frame}{Hipótesis}
  \begin{description}[H1]
    \item[H1] El estimador ponderado (A) es consistente y tiene el menor ECM cuando $n$ crece; B, C y D no convergen al valor verdadero.
    \item[H2] En 2026 la educación explica más desigualdad que cada una de las otras cuatro dimensiones.
    \item[H3] La desigualdad del ingreso laboral fue menor en 2026 que en 2018.
  \end{description}
  \vspace{1em}
  \small Todas las pruebas son unilaterales, con $\alpha = 0,05$.
\end{frame}

\section{Datos y método}

\begin{frame}{Datos: Gran Encuesta Integrada de Hogares (DANE)}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Enero a junio de 2018 a 2026; 2020 no entra (recolección telefónica).
        \item Ocupados de 15 años o más, remunerados, con ingreso positivo, en 24 departamentos comunes.
        \item No respuesta de ingreso: su peso se reparte entre quienes responden (\emph{marcar, no borrar}).
        \item Cinco dimensiones: educación, formalidad, estrato, región y sexo.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{block}{En cifras}
        \begin{itemize}
          \item 130.000 a 161.000 personas por año
          \item $\approx$ 21,6 millones de ocupados representados (2026)
          \item Factor de expansión: FEX\_C\_2011 hasta 2021, FEX\_C18 desde 2022
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cuatro estimadores del índice de Theil}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Por sustitución}
      \[ \hat T_A = \frac{1}{W}\sum_i w_i \frac{y_i}{\hat\mu_w}\ln\frac{y_i}{\hat\mu_w} \]
      \begin{itemize}
        \item \textbf{A:} con el factor de expansión $w_i$
        \item \textbf{B:} la misma fórmula con $w_i = 1$
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Lognormales:} si $\ln y \sim N(m,\sigma^2)$, entonces $T = \sigma^2/2$
      \begin{itemize}
        \item \textbf{C (momentos):} $\hat\sigma^2 = \ln(1 + CV^2)$
        \item \textbf{D (máx. verosimilitud):} $\hat\sigma^2 = \frac{1}{n}\sum_i (\ln y_i - \overline{\ln y})^2$
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1em}
  \begin{block}{Descomposición}
    $T = \underbrace{\textstyle\sum_k s_k \ln(\mu_k/\mu)}_{\text{entre grupos}} + \underbrace{\textstyle\sum_k s_k T_k}_{\text{dentro de los grupos}}$, \quad $s_k$ = participación del grupo en el ingreso.
  \end{block}
\end{frame}

\begin{frame}{Diseño del experimento y de la inferencia}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Simulación Monte Carlo (H1)}
        \begin{itemize}
          \item ``Población'': GEIH 2024 completa
          \item Valor de referencia: $T = 0,5043$ (A)
          \item 500 submuestras sin reemplazo para $n$ = 500, 2.000, 10.000 y 50.000
          \item Sesgo, varianza, ECM y eficiencia relativa
          \item Prueba $t$ pareada con $n = 50.000$
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Bootstrap (H2, H3)}
        \begin{itemize}
          \item 500 réplicas: personas con reemplazo, cada una con su factor
          \item IC del 95~\% del Theil de cada año
          \item H2: percentil 5 de $S^*_{edu} - S^*_k$
          \item H3: percentil 95 de $T^*_{2026} - T^*_{2018}$
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Resultados}

\begin{frame}{El ingreso laboral tiene un pico en el salario mínimo}

    \end{column}
    \begin{column}{0.38\textwidth}

      \begin{itemize}
        \item Media \$2,3 millones; mediana \$1,75 millones.
        \item 15,5~\% gana $\approx$ 1 salario mínimo; 39,5~\% gana menos.
        \item Muy asimétrico: asimetría 9,9.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{¿Es lognormal el ingreso? No}
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}
      \includegraphics[width=\textwidth]{fig_ln_ingreso_normal.pdf}
    \end{column}
    \begin{column}{0.38\textwidth}
      \small
      \begin{itemize}
        \item ln(ingreso) tiene colas más pesadas que la normal (curtosis en exceso 4,1).
        \item Pico en el salario mínimo.
        \item Anticipa que C y D, que suponen lognormalidad, fallarán.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Simulación: solo A es consistente}
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}
      \includegraphics[width=\textwidth]{fig_simulacion_ecm.pdf}
    \end{column}
    \begin{column}{0.38\textwidth}
      \small
      \begin{itemize}
        \item El ECM de A cae sin parar al crecer $n$.
        \item B y D se estancan: su sesgo no desaparece.
        \item Con $n = 500$, D tiene menor ECM (poca varianza); A gana desde $n = 10.000$.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Simulación con $n = 50.000$}
  \centering
  \begin{tabular}{@{}lrrr@{}}
    \toprule

    \midrule
    A (ponderado)            & 0,0003    & 0,25 & --- \\

  \begin{block}{Lectura}
    B y D subestiman la desigualdad en cerca de 15~\%; C la sobreestima en cerca de 5~\%. El ECM de A es entre 4 y 24 veces menor.
  \end{block}
\end{frame}

\begin{frame}{¿Qué explica la desigualdad?}
  \begin{columns}[c]
    \begin{column}{0.62\textwidth}

    \end{column}
    \begin{column}{0.38\textwidth}
      \small
      \begin{itemize}
        \item Estrato: 31--38~\% del Theil.
        \item Educación (24--27~\%) y formalidad (22--27~\%) casi empatadas.

\section{Conclusiones}

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item \textbf{Usar el estimador ponderado (A)} con la GEIH: es el único consistente. Ignorar el factor de expansión o suponer lognormalidad produce sesgos que no desaparecen.
    \item \textbf{La desigualdad} subió hasta 2022 y bajó hasta su mínimo en 2026, pero la caída frente a 2018 no es significativa.
    \item \textbf{El estrato} es lo que más explica la desigualdad; la educación y la formalidad empatan en segundo lugar.
  \end{itemize}
  \vspace{0.8em}
  \begin{block}{Limitaciones}
    \small El valor de referencia es el propio A sobre la GEIH 2024; el cambio de marco en 2022; la no respuesta es mayor en estratos altos; el bootstrap ignora los conglomerados.
  \end{block}
\end{frame}
