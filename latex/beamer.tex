\begin{frame}{La desigualdad se estima, no se observa}
  \begin{itemize}
    \setlength{\itemsep}{0.6em}
    \item Colombia es uno de los países más desiguales de América Latina.
    \item Las cifras de desigualdad salen de \textbf{encuestas}: son estimaciones.
    \item Con los mismos datos hay varias formas de estimar el índice de Theil. Si se elige mal, el resultado queda \textbf{sesgado}.
  \end{itemize}
  \vspace{1em}
  \begin{block}{Pregunta}
    De cuatro estimadores del índice de Theil, ¿cuál estima mejor la desigualdad del ingreso laboral con la GEIH? Con ese estimador: ¿cómo cambió la desigualdad entre 2018 y 2026, y qué la explica?
  \end{block}
\end{frame}

\begin{frame}{Hipótesis}
  \vspace{0.5em}
  \begin{description}[H1]
    \setlength{\itemsep}{0.8em}
    \item[H1] El estimador ponderado (A) es consistente y tiene el menor error cuando la muestra crece.
    \item[H2] En 2026 la educación explica más desigualdad que las otras cuatro dimensiones.
    \item[H3] La desigualdad fue menor en 2026 que en 2018.
  \end{description}
  \vspace{1.2em}
  {\color{gris}\small Pruebas unilaterales, $\alpha = 0,05$.}
\end{frame}

% ============================================================
\begin{frame}{Datos: Gran Encuesta Integrada de Hogares (DANE)}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \setlength{\itemsep}{0.5em}
        \item Enero a junio de 2018 a 2026. 2020 no entra: la encuesta fue telefónica.
        \item Ocupados de 15 años o más, remunerados, con ingreso positivo, en 24 departamentos.
        \item Regla: \emph{marcar, no borrar}. El peso de quien no reporta ingreso se reparte entre personas parecidas.
        \item Cinco dimensiones: educación, formalidad, estrato, región y sexo.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \begin{block}{En cifras}
        \begin{itemize}
          \item 130.000 a 161.000 personas por año
          \item 21,6 millones de ocupados representados
          \item Cambio de marco muestral en 2022
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cuatro formas de estimar el Theil}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Aplicar la fórmula a la muestra}
      \[ \hat T_A = \frac{1}{W}\sum_i w_i \frac{y_i}{\hat\mu_w}\ln\frac{y_i}{\hat\mu_w} \]
      \begin{itemize}
        \item \textbf{A:} con el factor de expansión $w_i$
        \item \textbf{B:} igual, pero con $w_i = 1$
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Suponer que el ingreso es lognormal}\\[0.4em]
      Si $\ln y \sim N(m,\sigma^2)$, entonces $T = \sigma^2/2$
      \begin{itemize}
        \item \textbf{C:} $\hat\sigma^2 = \ln(1 + CV^2)$ (momentos)
        \item \textbf{D:} $\hat\sigma^2 = \frac{1}{n}\sum_i (\ln y_i - \overline{\ln y})^2$ (máx. verosimilitud)
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1em}
  \begin{block}{Descomposición}
    $T = \underbrace{\textstyle\sum_k s_k \ln(\mu_k/\mu)}_{\text{entre grupos}} + \underbrace{\textstyle\sum_k s_k T_k}_{\text{dentro de los grupos}}$ \qquad $s_k$: parte del ingreso total del grupo $k$
  \end{block}
\end{frame}

\begin{frame}{Cómo lo hicimos}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Simulación (H1)}
        \begin{enumerate}
          \item GEIH 2024 completa como ``población'': $T = 0,5043$
          \item 500 submuestras de $n$ = 500, 2.000, 10.000 y 50.000
          \item A, B, C y D en cada una
          \item Sesgo, ECM y eficiencia relativa
          \item Prueba $t$ pareada con $n = 50.000$
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Bootstrap (H2, H3)}
        \begin{enumerate}
          \item Remuestrear personas con reemplazo, 500 veces
          \item Recalcular el Theil y su descomposición
          \item IC del 95~\%: percentiles 2,5 y 97,5
          \item H2 y H3 con percentiles de las diferencias
        \end{enumerate}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.8em}
  {\color{gris}\small Todo en R, con semillas fijas: \texttt{github.com/gabrielsalcedo/statistics-geih}}
\end{frame}

        \item Por eso esperábamos que \textbf{C y D fallaran}.

  \end{itemize}

\begin{frame}{Conclusiones}
  \begin{enumerate}
    \setlength{\itemsep}{0.7em}
    \item \textbf{Con la GEIH hay que usar el estimador ponderado (A).} Ignorar los pesos o suponer lognormalidad deja sesgos que no desaparecen.
    \item \textbf{La desigualdad} subió hasta 2022 y bajó hasta su mínimo en 2026, pero frente a 2018 la caída no es significativa.
    \item \textbf{El estrato} es lo que más explica la desigualdad.
  \end{enumerate}
  \vspace{0.8em}
  \begin{block}{Limitaciones}
    \small Referencia = A sobre la GEIH 2024 · cambio de marco en 2022 · más no respuesta en estratos altos · el bootstrap ignora los conglomerados.
  \end{block}
\end{frame}
